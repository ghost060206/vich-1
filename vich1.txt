CREATE OR REPLACE FUNCTION get_total_damage (
    p_boss_id IN NUMBER
) RETURN NUMBER IS
    v_multiplier   damage.multiplier%TYPE;
    v_damage       damage.damage%TYPE;
    v_attack_count attacks.attack_count%TYPE;
    v_total        NUMBER;
BEGIN
    SELECT d.multiplier, d.damage, a.attack_count
    INTO v_multiplier, v_damage, v_attack_count
    FROM boss b
    JOIN damage d  ON b.damage_id = d.id_d
    JOIN attacks a ON b.attacks_id = a.id_a
    WHERE b.id_b = p_boss_id;

    v_total := v_multiplier * v_damage * v_attack_count;
    RETURN v_total;
EXCEPTION
    WHEN NO_DATA_FOUND THEN
        RETURN NULL;
END;

// Интегрирование методом средних прямоугольников
double integrate(function<double(double)> f, double a, double x, int n) {
    if (x == a) return 0;
    double h = (x - a) / n;
    double sum = 0;
    for (int i = 0; i < n; ++i)
        sum += f(a + (i + 0.5) * h);
    return sum * h;
}

// Невязка
double F(function<double(double)> f, double a, double b, double x, int n) {
    return integrate(f, a, x, n) - b;
}

// Решение уравнения методом бисекции
double solve(function<double(double)> f, double a, double b, int n_int, double tol) {
    if (abs(b) < 1e-12) return a;

    // Поиск интервала
    double left = a, right = a;
    double step = 1.0;
    int dir = (b > 0) ? 1 : -1;
    while (true) {
        right = a + dir * step;
        if (F(f, a, b, left, n_int) * F(f, a, b, right, n_int) <= 0)
            break;
        step *= 2;
        left = right;
        if (abs(right - a) > 1e7)
            throw runtime_error("Интервал не найден");
    }
    if (left > right) swap(left, right);

    // Бисекция
    for (int i = 0; i < 100; ++i) {
        double mid = (left + right) / 2;
        double Fmid = F(f, a, b, mid, n_int);
        if (abs(Fmid) < tol || (right - left) / 2 < tol)
            return mid;
        if (F(f, a, b, left, n_int) * Fmid < 0)
            right = mid;
        else
            left = mid;
    }
    return (left + right) / 2;
}

// Класс для интегрирования методом прямоугольников
class RectangleIntegration {
    function<double(double)> f;
    double a;
    int n;
public:
    RectangleIntegration(function<double(double)> func, double lower, int intervals) 
        : f(func), a(lower), n(intervals) {}

    double leftRectangles(double x) {
        if (x == a) return 0;
        double h = (x - a) / n, sum = 0;
        for (int i = 0; i < n; ++i) sum += f(a + i * h);
        return sum * h;
    }

    double rightRectangles(double x) {
        if (x == a) return 0;
        double h = (x - a) / n, sum = 0;
        for (int i = 1; i <= n; ++i) sum += f(a + i * h);
        return sum * h;
    }

    double middleRectangles(double x) {
        if (x == a) return 0;
        double h = (x - a) / n, sum = 0;
        for (int i = 0; i < n; ++i) sum += f(a + (i + 0.5) * h);
        return sum * h;
    }
};

// Решатель уравнения ∫_a^x f(t) dt = b
class EquationSolver {
    function<double(double)> f;
    double a, b;
    int n_int;
    double tol;
    int max_iter;
    RectangleIntegration integrator;
    double (RectangleIntegration::*integralMethod)(double);

    double F(double x) {
        return (integrator.*integralMethod)(x) - b;
    }

    // Поиск интервала, содержащего корень
    void findInterval(double& left, double& right) {
        if (abs(b) < 1e-12) { left = right = a; return; }

        int dir = (b > 0) ? 1 : -1;
        double step = 1.0;
        left = a;
        while (true) {
            right = a + dir * step;
            if (F(left) * F(right) <= 0) {
                if (left > right) swap(left, right);
                return;
            }
            step *= 2;
            left = right;
            if (abs(right - a) > 1e7) 
                throw runtime_error("Не найден интервал с корнем");
        }
    }

public:
    EquationSolver(function<double(double)> func, double lower, double right,
                   string method = "middle", int intervals = 100,
                   double tolerance = 1e-6, int maxIter = 100)
        : f(func), a(lower), b(right), n_int(intervals), tol(tolerance),
          max_iter(maxIter), integrator(func, lower, intervals) {
        if (method == "left")
            integralMethod = &RectangleIntegration::leftRectangles;
        else if (method == "right")
            integralMethod = &RectangleIntegration::rightRectangles;
        else
            integralMethod = &RectangleIntegration::middleRectangles;
    }

    double solve(double left = -1, double right = -1) {
        if (abs(b) < 1e-12) return a;

        if (left == -1 || right == -1) {
            try { findInterval(left, right); }
            catch (const exception& e) { throw; }
        }

        double F_left = F(left), F_right = F(right);
        if (abs(F_left) < tol) return left;
        if (abs(F_right) < tol) return right;
        if (F_left * F_right > 0)
            throw runtime_error("Нет корня на интервале");

        for (int i = 0; i < max_iter; ++i) {
            double mid = (left + right) / 2, F_mid = F(mid);
            if (abs(F_mid) < tol || (right - left) / 2 < tol) return mid;
            if (F_left * F_mid < 0) { right = mid; F_right = F_mid; }
            else { left = mid; F_left = F_mid; }
        }
        return (left + right) / 2;
    }
};

void adam2_explicit(double (*f)(double,double), double y0, double t0, double tend, double h,
                    std::vector<double>& t, std::vector<double>& y) {
    int n = (int)((tend - t0)/h + 0.5);
    t.resize(n+1); y.resize(n+1);
    t[0]=t0; y[0]=y0;
    // старт явным Эйлером
    double t1=t0+h, y1=y0+h*f(t0,y0);
    t[1]=t1; y[1]=y1;
    double fp = f(t0,y0), fc = f(t1,y1);
    for (int i=1; i<n; ++i) {
        double tn = t[i] + h;
        double yn = y[i] + h/2.0 * (3.0*fc - fp);
        t[i+1]=tn; y[i+1]=yn;
        fp = fc;
        fc = f(tn, yn);
    }
}

void adam2_implicit(double (*f)(double,double), double y0, double t0, double tend, double h,
                    int maxit, double tol,
                    std::vector<double>& t, std::vector<double>& y) {
    int n = (int)((tend - t0)/h + 0.5);
    t.resize(n+1); y.resize(n+1);
    t[0]=t0; y[0]=y0;
    double t1=t0+h, y1=y0+h*f(t0,y0);
    t[1]=t1; y[1]=y1;
    double fp = f(t0,y0), fc = f(t1,y1);
    for (int i=1; i<n; ++i) {
        double tn = t[i] + h;
        double yn = y[i] + h/2.0 * (3.0*fc - fp); // прогноз
        for (int it=0; it<maxit; ++it) {
            double fn = f(tn, yn);
            double yn_new = y[i] + h/12.0 * (5.0*fn + 8.0*fc - fp);
            if (std::abs(yn_new - yn) < tol) { yn = yn_new; break; }
            yn = yn_new;
        }
        t[i+1]=tn; y[i+1]=yn;
        fp = fc;
        fc = f(tn, yn);
    }
}

    auto print = [](const std::vector<double>& t, const std::vector<double>& y) {
        std::cout << std::fixed << std::setprecision(6);
        std::cout << "t\tЧисл.\tТочн.\tОшибка\n";
        double maxe=0;
        for (size_t i=0; i<t.size(); ++i) {
            double ex = exact(t[i]);
            double e = std::abs(y[i]-ex);
            if(e>maxe) maxe=e;
            std::cout << t[i] << '\t' << y[i] << '\t' << ex << '\t' << e << '\n';
        }
        std::cout << "Max error: " << maxe << "\n\n";
    };
    
    std::cout << "=== Явный 2-шаговый Адамс ===\n";
    adam2_explicit(f, y0, t0, tend, h, t_exp, y_exp);
    print(t_exp, y_exp);
    
    std::cout << "=== Неявный 2-шаговый Адамс ===\n";
    adam2_implicit(f, y0, t0, tend, h, 20, 1e-10, t_imp, y_imp);
    print(t_imp, y_imp);
    
    return 0;
}
