// Интегрирование методом средних прямоугольников
double integrate(function<double(double)> f, double a, double x, int n) {
    if (x == a) return 0;
    double h = (x - a) / n;
    double sum = 0;
    for (int i = 0; i < n; ++i)
        sum += f(a + (i + 0.5) * h);
    return sum * h;
}

// Невязка
double F(function<double(double)> f, double a, double b, double x, int n) {
    return integrate(f, a, x, n) - b;
}

// Решение уравнения методом бисекции
double solve(function<double(double)> f, double a, double b, int n_int, double tol) {
    if (abs(b) < 1e-12) return a;

    // Поиск интервала
    double left = a, right = a;
    double step = 1.0;
    int dir = (b > 0) ? 1 : -1;
    while (true) {
        right = a + dir * step;
        if (F(f, a, b, left, n_int) * F(f, a, b, right, n_int) <= 0)
            break;
        step *= 2;
        left = right;
        if (abs(right - a) > 1e7)
            throw runtime_error("Интервал не найден");
    }
    if (left > right) swap(left, right);

    // Бисекция
    for (int i = 0; i < 100; ++i) {
        double mid = (left + right) / 2;
        double Fmid = F(f, a, b, mid, n_int);
        if (abs(Fmid) < tol || (right - left) / 2 < tol)
            return mid;
        if (F(f, a, b, left, n_int) * Fmid < 0)
            right = mid;
        else
            left = mid;
    }
    return (left + right) / 2;
}

// Класс для интегрирования методом прямоугольников
class RectangleIntegration {
    function<double(double)> f;
    double a;
    int n;
public:
    RectangleIntegration(function<double(double)> func, double lower, int intervals) 
        : f(func), a(lower), n(intervals) {}

    double leftRectangles(double x) {
        if (x == a) return 0;
        double h = (x - a) / n, sum = 0;
        for (int i = 0; i < n; ++i) sum += f(a + i * h);
        return sum * h;
    }

    double rightRectangles(double x) {
        if (x == a) return 0;
        double h = (x - a) / n, sum = 0;
        for (int i = 1; i <= n; ++i) sum += f(a + i * h);
        return sum * h;
    }

    double middleRectangles(double x) {
        if (x == a) return 0;
        double h = (x - a) / n, sum = 0;
        for (int i = 0; i < n; ++i) sum += f(a + (i + 0.5) * h);
        return sum * h;
    }
};

// Решатель уравнения ∫_a^x f(t) dt = b
class EquationSolver {
    function<double(double)> f;
    double a, b;
    int n_int;
    double tol;
    int max_iter;
    RectangleIntegration integrator;
    double (RectangleIntegration::*integralMethod)(double);

    double F(double x) {
        return (integrator.*integralMethod)(x) - b;
    }

    // Поиск интервала, содержащего корень
    void findInterval(double& left, double& right) {
        if (abs(b) < 1e-12) { left = right = a; return; }

        int dir = (b > 0) ? 1 : -1;
        double step = 1.0;
        left = a;
        while (true) {
            right = a + dir * step;
            if (F(left) * F(right) <= 0) {
                if (left > right) swap(left, right);
                return;
            }
            step *= 2;
            left = right;
            if (abs(right - a) > 1e7) 
                throw runtime_error("Не найден интервал с корнем");
        }
    }

public:
    EquationSolver(function<double(double)> func, double lower, double right,
                   string method = "middle", int intervals = 100,
                   double tolerance = 1e-6, int maxIter = 100)
        : f(func), a(lower), b(right), n_int(intervals), tol(tolerance),
          max_iter(maxIter), integrator(func, lower, intervals) {
        if (method == "left")
            integralMethod = &RectangleIntegration::leftRectangles;
        else if (method == "right")
            integralMethod = &RectangleIntegration::rightRectangles;
        else
            integralMethod = &RectangleIntegration::middleRectangles;
    }

    double solve(double left = -1, double right = -1) {
        if (abs(b) < 1e-12) return a;

        if (left == -1 || right == -1) {
            try { findInterval(left, right); }
            catch (const exception& e) { throw; }
        }

        double F_left = F(left), F_right = F(right);
        if (abs(F_left) < tol) return left;
        if (abs(F_right) < tol) return right;
        if (F_left * F_right > 0)
            throw runtime_error("Нет корня на интервале");

        for (int i = 0; i < max_iter; ++i) {
            double mid = (left + right) / 2, F_mid = F(mid);
            if (abs(F_mid) < tol || (right - left) / 2 < tol) return mid;
            if (F_left * F_mid < 0) { right = mid; F_right = F_mid; }
            else { left = mid; F_left = F_mid; }
        }
        return (left + right) / 2;
    }
};

// Неявный 2-шаговый метод Адамса (AM2)
void adam2_implicit(double y0, double t0, double tend, double h,
                    int maxit, double tol,
                    std::vector<double>& t, std::vector<double>& y) {
    
    int n = (int)((tend - t0) / h + 0.5);  // количество шагов
    t.resize(n + 1);
    y.resize(n + 1);
    
    // Начальное условие
    t[0] = t0;
    y[0] = y0;
    
    // Шаг 1: получаем y1 явным методом Эйлера (старт)
    double t1 = t0 + h;
    double y1 = y0 + h * f(t0, y0);
    t[1] = t1;
    y[1] = y1;
    
    // Значения производных на первых двух шагах
    double fp = f(t0, y0);   // f_{n-1}
    double fc = f(t1, y1);   // f_n
    
    // Основной цикл (начиная с n = 1, вычисляем y2, y3, ...)
    for (int i = 1; i < n; ++i) {
        double tn = t[i] + h;   // t_{n+1}
        
        // Предиктор (явный метод Адамса-Башфорт 2-го порядка)
        double yn = y[i] + h / 2.0 * (3.0 * fc - fp);
        
        // Корректор (итерации по неявной формуле AM2)
        for (int iter = 0; iter < maxit; ++iter) {
            double fn = f(tn, yn);
            double yn_new = y[i] + h / 12.0 * (5.0 * fn + 8.0 * fc - fp);
            
            if (std::abs(yn_new - yn) < tol) {
                yn = yn_new;
                break;
            }
            yn = yn_new;
        }
        
        // Сохраняем результат
        t[i + 1] = tn;
        y[i + 1] = yn;
        
        // Обновляем производные для следующего шага
        fp = fc;
        fc = f(tn, yn);
    }
}

        double tn = t[i] + h;
        
        // Предиктор (явный Адамс-Башфорт)
        double yn = y[i] + h / 2.0 * (3.0 * fc - fp);
        
        // Корректор (итерации)
        for (int iter = 0; iter < maxit; ++iter) {
            double fn = f(tn, yn);
            double yn_new = y[i] + h / 12.0 * (5.0 * fn + 8.0 * fc - fp);
            if (std::abs(yn_new - yn) < tol) {
                yn = yn_new;
                break;
            }
            yn = yn_new;
        }
        
        t[i + 1] = tn;
        y[i + 1] = yn;
        
        fp = fc;
        fc = f(tn, yn);
    }
}

int main() {
    // ==================== ПАРАМЕТРЫ ЗАДАЧИ ====================
    double t0 = 0.0;      // начальное время
    double tend = 1.0;    // конечное время
    double y0 = 1.0;      // начальное условие
    double h = 0.1;       // шаг интегрирования
    
    int maxit = 20;       // максимальное число итераций
    double tol = 1e-10;   // точность итераций
    // ===========================================================
    
    std::vector<double> t, y;
    adam2_implicit(y0, t0, tend, h, maxit, tol, t, y);
    
    // ========== ВЫВОД В CSV ДЛЯ EXCEL ==========
    std::ofstream out("results.csv");
    if (!out.is_open()) {
        std::cerr << "Ошибка: не удалось создать файл results.csv" << std::endl;
        return 1;
    }
    
    // Заголовки столбцов (для Excel)
    out << "t;Численное решение;Точное решение\n";
    
    for (size_t i = 0; i < t.size(); ++i) {
        out << t[i] << ";" << y[i] << ";" << exact(t[i]) << "\n";
    }
    out.close();
    // ============================================
    
    // ========== ВЫВОД ТАБЛИЦЫ В КОНСОЛЬ ==========
    std::cout << std::fixed << std::setprecision(8);
    std::cout << "t\t\tЧисленное\tТочное\t\tОшибка\n";
    double max_err = 0.0;
    
    for (size_t i = 0; i < t.size(); ++i) {
        double ex = exact(t[i]);
        double err = std::abs(y[i] - ex);
        if (err > max_err) max_err = err;
        std::cout << t[i] << "\t\t" << y[i] << "\t\t" << ex << "\t\t" << err << "\n";
    }
    
    std::cout << "\nМаксимальная ошибка: " << max_err << std::endl;
    std::cout << "\nДанные для Excel сохранены в файл 'results.csv'" << std::endl;
    std::cout << "Откройте файл в Excel через меню Данные -> Из текстового файла, выберите разделитель ';'" << std::endl;
    
    return 0;
}
