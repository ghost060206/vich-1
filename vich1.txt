  Найти оригинал Лапласа функции \(F(p) = \frac{1}{(p-1)^2(p^2+1)}\) и построить его график.
  
  ### Решение в Maple:
  ```maple
  restart;
  with(inttrans):
  F17 := 1/((p-1)^2*(p^2+1));
  f17 := invlaplace(F17, p, t);
  simplify(f17);
  plot(f17, t=0..10);
  ```

#include <iostream>
#include <vector>
#include <fstream>

// Решение трёхдиагональной системы методом прогонки
vector<double> solveTridiagonal(const vector<double>& a, const vector<double>& b,
                                const vector<double>& c, const vector<double>& d) {
    int n = d.size();
    vector<double> P(n - 1), Q(n - 1), x(n);
    P[0] = -c[0] / b[0];
    Q[0] = d[0] / b[0];
    for (int i = 1; i < n - 1; i++) {
        double denom = b[i] + a[i] * P[i - 1];
        P[i] = -c[i] / denom;
        Q[i] = (d[i] - a[i] * Q[i - 1]) / denom;
    }
    x[n - 1] = (d[n - 1] - a[n - 1] * Q[n - 2]) / (b[n - 1] + a[n - 1] * P[n - 2]);
    for (int i = n - 2; i >= 0; i--) x[i] = P[i] * x[i + 1] + Q[i];
    return x;
}

// Класс кубического сплайна
class CubicSpline {
private:
    vector<double> x, a, b, c, d;

public:
    CubicSpline(const vector<double>& x_nodes, const vector<double>& y_nodes) : x(x_nodes), a(y_nodes) {
        int n = x.size();
        vector<double> h(n - 1);
        for (int i = 0; i < n - 1; i++) h[i] = x[i + 1] - x[i];

        int m = n - 2;
        if (m > 0) {
            vector<double> a_sub(m), b_diag(m), c_super(m), d_rhs(m);
            for (int i = 1; i <= m; i++) {
                double rhs = 3.0 * ((a[i + 1] - a[i]) / h[i] - (a[i] - a[i - 1]) / h[i - 1]);
                a_sub[i - 1] = h[i - 1];
                b_diag[i - 1] = 2.0 * (h[i - 1] + h[i]);
                c_super[i - 1] = h[i];
                d_rhs[i - 1] = rhs;
            }
            vector<double> c_inner = solveTridiagonal(a_sub, b_diag, c_super, d_rhs);
            c.assign(n, 0.0);
            for (int i = 0; i < m; i++) c[i + 1] = c_inner[i];
        } else {
            c.assign(n, 0.0);
        }

        b.resize(n - 1);
        d.resize(n - 1);
        for (int i = 0; i < n - 1; i++) {
            d[i] = (c[i + 1] - c[i]) / (3.0 * h[i]);
            b[i] = (a[i + 1] - a[i]) / h[i] - h[i] * (c[i + 1] + 2.0 * c[i]) / 3.0;
        }
    }

        int i = 0;
        while (i < (int)x.size() - 1 && x_val > x[i + 1]) i++;
        double dx = x_val - x[i];

Понял, проблема в том, что при копировании математических формул из моего ответа символы могут вставляться каждый с новой строки. Вот формулы в виде обычного текста, который копируется нормально:

Математические формулы (текстовый вариант)
1. Представление сплайна на отрезке
На каждом отрезке [x_i, x_{i+1}] сплайн имеет вид:

S_i(x) = a_i + b_i(x - x_i) + c_i(x - x_i)^2 + d_i*(x - x_i)^3

где:
a_i = y_i — значение функции в левом узле
h_i = x_{i+1} - x_i — длина отрезка

2. Выражение коэффициентов через c_i
Из условий непрерывности второй производной:

d_i = (c_{i+1} - c_i) / (3 * h_i)

Из условия прохождения через правый узел S_i(x_{i+1}) = y_{i+1}:

b_i = (y_{i+1} - y_i) / h_i - (h_i / 3) * (2*c_i + c_{i+1})

3. Система уравнений для c_i
Из условия непрерывности первой производной S'i(x{i+1}) = S'{i+1}(x{i+1}) получаем для внутренних узлов i = 1, 2, ..., n-1:

h_{i-1} * c_{i-1} + 2*(h_{i-1} + h_i) * c_i + h_i * c_{i+1} = 3 * [ (y_{i+1} - y_i)/h_i - (y_i - y_{i-1})/h_{i-1} ]

4. Краевые условия (естественный сплайн)
Вторая производная на концах равна нулю:

c_0 = 0
c_n = 0

5. Трёхдиагональная система в матричной форме
Для n узлов система имеет вид:

[ 2(h0+h1) h1 0 ... 0 ] [c1] = [ 3*((y2-y1)/h1 - (y1-y0)/h0) ]
[ h1 2(h1+h2) h2 ... 0 ] [c2] [ 3*((y3-y2)/h2 - (y2-y1)/h1) ]
[ 0 h2 2(h2+h3) ... 0 ] [c3] [ ... ]
[ ... ... ... ... h_{n-2}] [..] [ ... ]
[ 0 0 0 h_{n-2} 2(h_{n-2}+h_{n-1})] [c_{n-1}] = [ 3*((y_n - y_{n-1})/h_{n-1} - (y_{n-1} - y_{n-2})/h_{n-2}) ]

6. Метод прогонки (Thomas algorithm)
Для решения трёхдиагональной системы:
a_i * x_{i-1} + b_i * x_i + c_i * x_{i+1} = d_i, i = 0, 1, ..., n-1

Прямой ход (вычисление прогоночных коэффициентов):

P0 = -c0 / b0
Q0 = d0 / b0

Pi = -ci / (bi + ai * P_{i-1}), i = 1, 2, ..., n-2
Qi = (di - ai * Q_{i-1}) / (bi + ai * P_{i-1}), i = 1, 2, ..., n-2

Обратный ход:

x_{n-1} = (d_{n-1} - a_{n-1} * Q_{n-2}) / (b_{n-1} + a_{n-1} * P_{n-2})

x_i = Pi * x_{i+1} + Qi, i = n-2, n-3, ..., 0








#include <iostream>
#include <cmath>
#include <iomanip>

using namespace std;

// Исходная функция f(x). Здесь пример: f(x) = x^2 - 2  (корень sqrt(2))
double f(double x) {
    return x * x - 2;
}

// Итерационная функция метода простой итерации: g(x) = x - f(x)
double g(double x) {
    return x - f(x);
}

// Метод Вегстейна для уточнения корня
double wegstein(double x0, double eps, int maxIter, bool& converged) {
    double x = x0;          // текущее приближение
    double xPrev;           // предыдущее приближение (для проверки сходимости)
    int iter = 0;

    converged = true;
    cout << "Начальное приближение: x0 = " << x0 << endl;

    do {
        double y = g(x);          // g(x_n)
        double z = g(y);          // g(g(x_n))
        double denom = z - 2 * y + x;

        // Защита от деления на ноль
        if (fabs(denom) < 1e-12) {
            cerr << "Знаменатель близок к нулю. Используется шаг простой итерации." << endl;
            xPrev = x;
            x = y;                 // антикризисный шаг
        } else {
            xPrev = x;
            x = x - (y - x) * (y - x) / denom;   // формула Вегстейна
        }

        iter++;
        cout << "Итерация " << iter << ": x = " << setprecision(12) << x
             << ", f(x) = " << f(x) << endl;

        if (iter > maxIter) {
            converged = false;
            break;
        }
    } while (fabs(x - xPrev) > eps);   // условие остановки по разности приближений

    return x;
}

int main() {
    double x0 = 1.5;            // начальное приближение
    double eps = 1e-12;         // требуемая точность
    int maxIter = 50;           // максимальное число итераций
    bool converged;

    double root = wegstein(x0, eps, maxIter, converged);

    if (converged) {
        cout << "\nНайден корень: x = " << setprecision(12) << root << endl;
        cout << "Значение функции в корне: f(x) = " << f(root) << endl;
    } else {
        cout << "\nМетод не сошелся за " << maxIter << " итераций." << endl;
    }

    return 0;
}

// Класс для интегрирования методом прямоугольников
class RectangleIntegration {
    function<double(double)> f;
    double a;
    int n;
public:
    RectangleIntegration(function<double(double)> func, double lower, int intervals) 
        : f(func), a(lower), n(intervals) {}

    double leftRectangles(double x) {
        if (x == a) return 0;
        double h = (x - a) / n, sum = 0;
        for (int i = 0; i < n; ++i) sum += f(a + i * h);
        return sum * h;
    }

    double rightRectangles(double x) {
        if (x == a) return 0;
        double h = (x - a) / n, sum = 0;
        for (int i = 1; i <= n; ++i) sum += f(a + i * h);
        return sum * h;
    }

    double middleRectangles(double x) {
        if (x == a) return 0;
        double h = (x - a) / n, sum = 0;
        for (int i = 0; i < n; ++i) sum += f(a + (i + 0.5) * h);
        return sum * h;
    }
};

// Решатель уравнения ∫_a^x f(t) dt = b
class EquationSolver {
    function<double(double)> f;
    double a, b;
    int n_int;
    double tol;
    int max_iter;
    RectangleIntegration integrator;
    double (RectangleIntegration::*integralMethod)(double);

    double F(double x) {
        return (integrator.*integralMethod)(x) - b;
    }

    // Поиск интервала, содержащего корень
    void findInterval(double& left, double& right) {
        if (abs(b) < 1e-12) { left = right = a; return; }

        int dir = (b > 0) ? 1 : -1;
        double step = 1.0;
        left = a;
        while (true) {
            right = a + dir * step;
            if (F(left) * F(right) <= 0) {
                if (left > right) swap(left, right);
                return;
            }
            step *= 2;
            left = right;
            if (abs(right - a) > 1e7) 
                throw runtime_error("Не найден интервал с корнем");
        }
    }

public:
    EquationSolver(function<double(double)> func, double lower, double right,
                   string method = "middle", int intervals = 100,
                   double tolerance = 1e-6, int maxIter = 100)
        : f(func), a(lower), b(right), n_int(intervals), tol(tolerance),
          max_iter(maxIter), integrator(func, lower, intervals) {
        if (method == "left")
            integralMethod = &RectangleIntegration::leftRectangles;
        else if (method == "right")
            integralMethod = &RectangleIntegration::rightRectangles;
        else
            integralMethod = &RectangleIntegration::middleRectangles;
    }

    double solve(double left = -1, double right = -1) {
        if (abs(b) < 1e-12) return a;

        if (left == -1 || right == -1) {
            try { findInterval(left, right); }
            catch (const exception& e) { throw; }
        }

        double F_left = F(left), F_right = F(right);
        if (abs(F_left) < tol) return left;
        if (abs(F_right) < tol) return right;
        if (F_left * F_right > 0)
            throw runtime_error("Нет корня на интервале");

        for (int i = 0; i < max_iter; ++i) {
            double mid = (left + right) / 2, F_mid = F(mid);
            if (abs(F_mid) < tol || (right - left) / 2 < tol) return mid;
            if (F_left * F_mid < 0) { right = mid; F_right = F_mid; }
            else { left = mid; F_left = F_mid; }
        }
        return (left + right) / 2;
    }
};

// Тестовые функции
double f1(double x) { return x * x; }
double f2(double x) { return sin(x); }
double f3(double x) { return exp(x); }
double f4(double x) { return abs(x) < 1e-10 ? 1e10 : 1.0 / x; }

int main() {
    cout << fixed << setprecision(10);
    cout << "========================================================\n"
         << "Решение ∫_a^x f(t) dt = b методом прямоугольников (10а)\n"
         << "========================================================\n";

    // Демонстрация на трёх примерах
    struct Test { function<double(double)> f; double a, b; string name; double exact; };
    Test tests[] = {
        {f1, 0, 10, "x^2", 3.107232505953859},
        {f2, 0, 1,  "sin(x)", 1.5707963267948966},
        {f3, 0, 2,  "exp(x)", 1.0986122886681098}
    };

    string methods[] = {"left", "right", "middle"};
    string mnames[] = {"левых", "правых", "средних"};

    for (auto& t : tests) {
        cout << "\nПример: f(x) = " << t.name << ", a = " << t.a << ", b = " << t.b
             << "\nТочное решение: " << t.exact << "\n";
        for (int i = 0; i < 3; ++i) {
            EquationSolver solver(t.f, t.a, t.b, methods[i], 1000);
            double x = solver.solve();
            cout << mnames[i] << " прямоугольники: x = " << x
                 << ", погрешность = " << scientific << abs(x - t.exact) << fixed << "\n";
        }
    }

    // Интерактивный режим
    cout << "\n========================================================\n"
         << "Интерактивный режим\n"
         << "========================================================\n";
    int choice;
    do {
        cout << "\nВыберите функцию:\n"
             << "1: x^2   2: sin(x)   3: exp(x)   4: 1/x   0: выход\n";
        cin >> choice;
        if (choice == 0) break;
        if (choice < 1 || choice > 4) { cout << "Неверный выбор\n"; continue; }

        function<double(double)> f;
        string fname;
        switch (choice) {
            case 1: f = f1; fname = "x^2"; break;
            case 2: f = f2; fname = "sin(x)"; break;
            case 3: f = f3; fname = "exp(x)"; break;
            case 4: f = f4; fname = "1/x"; break;
        }

        double a, b_val;
        cout << "a = "; cin >> a;
        cout << "b = "; cin >> b_val;

        cout << "Метод (1-левые, 2-правые, 3-средние): ";
        int m; cin >> m;
        string method = (m == 1) ? "left" : (m == 2) ? "right" : "middle";
        string mname = (m == 1) ? "левых" : (m == 2) ? "правых" : "средних";

        int n_int; double tol;
        cout << "Число разбиений: "; cin >> n_int;
        cout << "Точность: "; cin >> tol;

        try {
            EquationSolver solver(f, a, b_val, method, n_int, tol);
            double x = solver.solve();
            cout << "\nРезультат:\n"
                 << "f(x) = " << fname << ", ∫_" << a << "^x " << fname << " dt = " << b_val << "\n"
                 << "Метод " << mname << " прямоугольников, разбиений = " << n_int << "\n"
                 << "x = " << x << "\n";

            // Проверка
            RectangleIntegration check(f, a, n_int);
            double I = (m == 1) ? check.leftRectangles(x) :
                        (m == 2) ? check.rightRectangles(x) :
                                   check.middleRectangles(x);
            cout << "Проверка: интеграл = " << I << ", невязка = " << scientific << abs(I - b_val) << "\n";
        } catch (const exception& e) {
            cout << "Ошибка: " << e.what() << "\n";
        }
    } while (true);

    cout << "Программа завершена.\n";
    return 0;
}
