#include <iostream>
#include <cmath>
#include <iomanip>

using namespace std;

// Пример уравнения: x^2 - 2 = 0  (корень sqrt(2) ≈ 1.414213562)
double f(double x) {
    return x * x - 2;
}

// Итерационная функция метода простой итерации
double g(double x) {
    return x - f(x);
}

// Метод Вегстейна
double wegstein(double x0, double eps, int maxIter, bool &converged) {
    double x = x0;
    double xPrev;
    int iter = 0;
    converged = true;

    cout << fixed << setprecision(12);
    cout << "Начальное приближение: x0 = " << x0 << endl;

    do {
        double y = g(x);           // g(x_n)
        double z = g(y);           // g(g(x_n))

        double denom = z - 2*y + x;

        if (fabs(denom) < 1e-12) {
            // Если знаменатель близок к нулю, делаем шаг простой итерации
            xPrev = x;
            x = y;
            cout << "Итерация " << iter+1 << " (simple): x = " << x 
                 << ", f(x) = " << f(x) << endl;
        } else {
            xPrev = x;
            x = x - (y - x)*(y - x) / denom;
            cout << "Итерация " << iter+1 << ": x = " << x 
                 << ", f(x) = " << f(x) << endl;
        }

        iter++;
        if (iter >= maxIter) {
            converged = false;
            break;
        }
    } while (fabs(x - xPrev) > eps);

    return x;
}

int main() {
    double x0 = 1.5;          // начальное приближение
    double eps = 1e-12;       // точность по разности приближений
    int maxIter = 50;         // максимум итераций
    bool converged;

    double root = wegstein(x0, eps, maxIter, converged);

    if (converged) {
        cout << "\nКорень найден: x = " << setprecision(12) << root << endl;
        cout << "f(x) = " << f(root) << endl;
    } else {
        cout << "\nМетод не сошёлся за " << maxIter << " итераций." << endl;
    }

    return 0;
}

// Класс для интегрирования методом прямоугольников
class RectangleIntegration {
    function<double(double)> f;
    double a;
    int n;
public:
    RectangleIntegration(function<double(double)> func, double lower, int intervals) 
        : f(func), a(lower), n(intervals) {}

    double leftRectangles(double x) {
        if (x == a) return 0;
        double h = (x - a) / n, sum = 0;
        for (int i = 0; i < n; ++i) sum += f(a + i * h);
        return sum * h;
    }

    double rightRectangles(double x) {
        if (x == a) return 0;
        double h = (x - a) / n, sum = 0;
        for (int i = 1; i <= n; ++i) sum += f(a + i * h);
        return sum * h;
    }

    double middleRectangles(double x) {
        if (x == a) return 0;
        double h = (x - a) / n, sum = 0;
        for (int i = 0; i < n; ++i) sum += f(a + (i + 0.5) * h);
        return sum * h;
    }
};

// Решатель уравнения ∫_a^x f(t) dt = b
class EquationSolver {
    function<double(double)> f;
    double a, b;
    int n_int;
    double tol;
    int max_iter;
    RectangleIntegration integrator;
    double (RectangleIntegration::*integralMethod)(double);

    double F(double x) {
        return (integrator.*integralMethod)(x) - b;
    }

    // Поиск интервала, содержащего корень
    void findInterval(double& left, double& right) {
        if (abs(b) < 1e-12) { left = right = a; return; }

        int dir = (b > 0) ? 1 : -1;
        double step = 1.0;
        left = a;
        while (true) {
            right = a + dir * step;
            if (F(left) * F(right) <= 0) {
                if (left > right) swap(left, right);
                return;
            }
            step *= 2;
            left = right;
            if (abs(right - a) > 1e7) 
                throw runtime_error("Не найден интервал с корнем");
        }
    }

public:
    EquationSolver(function<double(double)> func, double lower, double right,
                   string method = "middle", int intervals = 100,
                   double tolerance = 1e-6, int maxIter = 100)
        : f(func), a(lower), b(right), n_int(intervals), tol(tolerance),
          max_iter(maxIter), integrator(func, lower, intervals) {
        if (method == "left")
            integralMethod = &RectangleIntegration::leftRectangles;
        else if (method == "right")
            integralMethod = &RectangleIntegration::rightRectangles;
        else
            integralMethod = &RectangleIntegration::middleRectangles;
    }

    double solve(double left = -1, double right = -1) {
        if (abs(b) < 1e-12) return a;

        if (left == -1 || right == -1) {
            try { findInterval(left, right); }
            catch (const exception& e) { throw; }
        }

        double F_left = F(left), F_right = F(right);
        if (abs(F_left) < tol) return left;
        if (abs(F_right) < tol) return right;
        if (F_left * F_right > 0)
            throw runtime_error("Нет корня на интервале");

        for (int i = 0; i < max_iter; ++i) {
            double mid = (left + right) / 2, F_mid = F(mid);
            if (abs(F_mid) < tol || (right - left) / 2 < tol) return mid;
            if (F_left * F_mid < 0) { right = mid; F_right = F_mid; }
            else { left = mid; F_left = F_mid; }
        }
        return (left + right) / 2;
    }
};

#include <iostream>
#include <vector>
#include <cmath>
#include <iomanip>

double f(double t, double y) { return y; }
double exact(double t) { return std::exp(t); }

void adam2_explicit(double (*f)(double,double), double y0, double t0, double tend, double h,
                    std::vector<double>& t, std::vector<double>& y) {
    int n = (int)((tend - t0)/h + 0.5);
    t.resize(n+1); y.resize(n+1);
    t[0]=t0; y[0]=y0;
    // старт явным Эйлером
    double t1=t0+h, y1=y0+h*f(t0,y0);
    t[1]=t1; y[1]=y1;
    double fp = f(t0,y0), fc = f(t1,y1);
    for (int i=1; i<n; ++i) {
        double tn = t[i] + h;
        double yn = y[i] + h/2.0 * (3.0*fc - fp);
        t[i+1]=tn; y[i+1]=yn;
        fp = fc;
        fc = f(tn, yn);
    }
}

void adam2_implicit(double (*f)(double,double), double y0, double t0, double tend, double h,
                    int maxit, double tol,
                    std::vector<double>& t, std::vector<double>& y) {
    int n = (int)((tend - t0)/h + 0.5);
    t.resize(n+1); y.resize(n+1);
    t[0]=t0; y[0]=y0;
    double t1=t0+h, y1=y0+h*f(t0,y0);
    t[1]=t1; y[1]=y1;
    double fp = f(t0,y0), fc = f(t1,y1);
    for (int i=1; i<n; ++i) {
        double tn = t[i] + h;
        double yn = y[i] + h/2.0 * (3.0*fc - fp); // прогноз
        for (int it=0; it<maxit; ++it) {
            double fn = f(tn, yn);
            double yn_new = y[i] + h/12.0 * (5.0*fn + 8.0*fc - fp);
            if (std::abs(yn_new - yn) < tol) { yn = yn_new; break; }
            yn = yn_new;
        }
        t[i+1]=tn; y[i+1]=yn;
        fp = fc;
        fc = f(tn, yn);
    }
}
